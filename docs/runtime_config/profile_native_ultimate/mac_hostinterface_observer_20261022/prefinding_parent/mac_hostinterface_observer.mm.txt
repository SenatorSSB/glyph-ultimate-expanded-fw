void State::finish()noexcept{
 if(finished)return;finished=true;exit_code=engine.success&&healthy?0:2;
 // Uncertain completion retains EVERY referenced resource and journal to _Exit.
 if(engine.phase!=Phase::Uncertain&&!engine.pending()){
  // Queue cleanup AFTER the current completion block has returned. Keep data,
  // pipe and interface alive for the entire callback, including its epilogue.
  dispatch_async(queue,^{
   if(timer){dispatch_source_cancel(timer);timer=nullptr;}
   @try{pipe=nil;[host destroy];host=nil;data=nil;cleaned=true;}@catch(NSException*){exit_code=2;}
   record_end();
   if(journal){if(fclose(journal))exit_code=2;journal=nullptr;}
   dispatch_semaphore_signal(ended);
  });
 }else{record_end();dispatch_semaphore_signal(ended);}
}
